cmake_minimum_required(VERSION 3.16)
project(PicoDebugger VERSION 0.9.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PICODBG_GUI "Build the Qt Widgets desktop application (requires Qt development files)" ON)
option(PICODBG_SANITIZERS "Enable address and undefined behaviour sanitizers" OFF)
set(PICODBG_QT_MAJOR "AUTO" CACHE STRING "Qt major version: AUTO, 5 or 6")
set_property(CACHE PICODBG_QT_MAJOR PROPERTY STRINGS AUTO 5 6)

add_library(picodbg_core
    src/core/chip.cpp src/core/board.cpp src/core/memory.cpp src/core/elf.cpp
    src/core/thumb.cpp src/core/cpu.cpp src/core/debugger.cpp src/core/target.cpp
    src/core/timer.cpp src/core/peripherals.cpp src/core/peripheral_bus.cpp
    src/core/serial.cpp src/core/pwm.cpp
    src/core/registers/catalog_details.cpp src/core/registers/adc.cpp
    src/core/registers/i2c.cpp src/core/registers/dma.cpp src/core/registers/pio.cpp
    src/core/registers/serial_fields.cpp)
target_include_directories(picodbg_core PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(picodbg_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(PICODBG_SANITIZERS)
        target_compile_options(picodbg_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(picodbg_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()

if(PICODBG_GUI)
    if(PICODBG_QT_MAJOR STREQUAL "AUTO")
        find_package(QT NAMES Qt6 Qt5 COMPONENTS Widgets QUIET)
    elseif(PICODBG_QT_MAJOR MATCHES "^[56]$")
        find_package(Qt${PICODBG_QT_MAJOR} REQUIRED COMPONENTS Widgets)
        set(QT_FOUND TRUE)
        set(QT_VERSION_MAJOR ${PICODBG_QT_MAJOR})
    else()
        message(FATAL_ERROR "PICODBG_QT_MAJOR must be AUTO, 5 or 6")
    endif()
    if(QT_FOUND)
        if(QT_VERSION_MAJOR EQUAL 5)
            find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets)
        else()
            find_package(Qt6 REQUIRED COMPONENTS Widgets)
        endif()
        add_library(picodbg_widgets src/qt/main_window.cpp src/qt/window_theme.cpp src/qt/source_map.cpp src/qt/peripheral_panel.cpp src/qt/choice_box.cpp
            src/qt/register_inspector.cpp src/qt/inspection_workspace.cpp src/qt/peripheral_inspection.cpp
            src/qt/gdb_mi.cpp src/qt/gdb_target.cpp src/qt/probe_dialog.cpp
            src/qt/probe_serial.cpp src/qt/probe_serial_monitor.cpp
            src/qt/board_panel.cpp src/qt/uart_terminal.cpp src/qt/peripheral_explorer.cpp src/qt/virtual_time_panel.cpp
            src/qt/ide/ide_window.cpp src/qt/ide/project.cpp src/qt/ide/language_service.cpp
            src/qt/ide/code_editor.cpp src/qt/ide/git_panel.cpp src/qt/ide/toolchain_catalog.cpp
            src/qt/ide/text_search.cpp src/qt/ide/text_tools.cpp src/qt/ide/doxygen_dialog.cpp
            src/qt/ide/ide_templates.qrc src/qt/ide/ide_tools_resources.qrc)
        set_target_properties(picodbg_widgets PROPERTIES AUTORCC ON)
        target_include_directories(picodbg_widgets PUBLIC src/qt)
        target_link_libraries(picodbg_widgets PUBLIC picodbg_core Qt${QT_VERSION_MAJOR}::Widgets)
        find_package(Qt${QT_VERSION_MAJOR} QUIET COMPONENTS SerialPort)
        if(TARGET Qt${QT_VERSION_MAJOR}::SerialPort)
            target_link_libraries(picodbg_widgets PRIVATE Qt${QT_VERSION_MAJOR}::SerialPort)
            target_compile_definitions(picodbg_widgets PRIVATE PICODBG_QT_SERIALPORT)
        endif()
        if(QT_VERSION_MAJOR EQUAL 6)
            target_compile_features(picodbg_widgets PUBLIC cxx_std_17)
        endif()
        add_executable(pico-debugger src/qt/main.cpp src/qt/qt_diagnostics.cpp)
        target_compile_definitions(pico-debugger PRIVATE PICODBG_VERSION="${PROJECT_VERSION}")
        target_link_libraries(pico-debugger PRIVATE picodbg_widgets)
    else()
        message(FATAL_ERROR "Qt Widgets development files absent. Install qtbase5-dev or qt6-base-dev, then configure again. For an explicit core-only build use -DPICODBG_GUI=OFF.")
    endif()
endif()

    foreach(suite memory arithmetic control load_store exceptions decoder debugger elf)

    if(TARGET pico-debugger)
        add_test(NAME qt_smoke COMMAND pico-debugger --demo --smoke-test)
        set_tests_properties(qt_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
        add_executable(picodbg_gui_tests tests/test_gui.cpp)
        target_link_libraries(picodbg_gui_tests PRIVATE picodbg_widgets)
        add_test(NAME qt_interactions COMMAND picodbg_gui_tests)
        set_tests_properties(qt_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
        add_executable(picodbg_session_gui_tests tests/test_session_gui.cpp)

        add_test(NAME qt_session COMMAND picodbg_session_gui_tests)
        set_tests_properties(qt_session PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            add_executable(picodbg_probe_serial_tests tests/test_probe_serial.cpp)
            target_link_libraries(picodbg_probe_serial_tests PRIVATE picodbg_widgets)
            add_test(NAME qt_probe_serial COMMAND picodbg_probe_serial_tests)
            set_tests_properties(qt_probe_serial PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
        endif()
        if(Python3_Interpreter_FOUND AND UNIX)
            add_executable(picodbg_probe_tests tests/test_probe.cpp)
            target_link_libraries(picodbg_probe_tests PRIVATE picodbg_widgets)
            add_test(NAME probe_protocol_and_gui COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_probe_tests.py $<TARGET_FILE:picodbg_probe_tests>)
            set_tests_properties(probe_protocol_and_gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
            add_executable(picodbg_ide_tests tests/test_ide.cpp)
            target_link_libraries(picodbg_ide_tests PRIVATE picodbg_widgets)
            add_test(NAME qt_ide COMMAND picodbg_ide_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fake_clangd.py)
            set_tests_properties(qt_ide PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
            if(Python3_VERSION VERSION_GREATER_EQUAL "3.9")

        if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND QT_VERSION_MAJOR EQUAL 5)
            add_executable(picodbg_popup_tests tests/test_popup_activation.cpp)
            target_link_libraries(picodbg_popup_tests PRIVATE picodbg_widgets ${CMAKE_DL_LIBS})
            add_test(NAME qt_popup_activation COMMAND picodbg_popup_tests)
            set_tests_properties(qt_popup_activation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
        endif()
    endif()

    install(TARGETS pico-debugger RUNTIME DESTINATION bin)
