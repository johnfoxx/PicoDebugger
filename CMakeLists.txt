cmake_minimum_required(VERSION 3.16)
project(PicoDebugger VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PICODBG_GUI "Build the Qt Widgets desktop application (requires Qt development files)" ON)
option(PICODBG_SANITIZERS "Enable address and undefined behaviour sanitizers" OFF)

add_library(picodbg_core
    src/core/chip.cpp src/core/memory.cpp src/core/elf.cpp
    src/core/thumb.cpp src/core/cpu.cpp src/core/debugger.cpp src/core/timer.cpp src/core/peripherals.cpp)
target_include_directories(picodbg_core PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(picodbg_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(PICODBG_SANITIZERS)
        target_compile_options(picodbg_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(picodbg_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()

add_executable(pico-debugger-cli src/cli/main.cpp)
target_link_libraries(pico-debugger-cli PRIVATE picodbg_core)

if(PICODBG_GUI)
    find_package(QT NAMES Qt6 Qt5 COMPONENTS Widgets QUIET)
    if(QT_FOUND)
        find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
        add_library(picodbg_widgets src/qt/main_window.cpp src/qt/peripheral_panel.cpp)
        target_include_directories(picodbg_widgets PUBLIC src/qt)
        target_link_libraries(picodbg_widgets PUBLIC picodbg_core Qt${QT_VERSION_MAJOR}::Widgets)
        if(QT_VERSION_MAJOR EQUAL 6)
            target_compile_features(picodbg_widgets PUBLIC cxx_std_17)
        endif()
        add_executable(pico-debugger src/qt/main.cpp)
        target_link_libraries(pico-debugger PRIVATE picodbg_widgets)
    else()
        message(FATAL_ERROR "Qt Widgets development files absent. Install qtbase5-dev or qt6-base-dev, then configure again. For an explicit core/CLI-only build use -DPICODBG_GUI=OFF.")
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(picodbg_tests tests/test_main.cpp)
    target_link_libraries(picodbg_tests PRIVATE picodbg_core)
    foreach(suite memory arithmetic control load_store exceptions decoder debugger elf)
        add_test(NAME ${suite} COMMAND picodbg_tests ${suite})
        set_tests_properties(${suite} PROPERTIES TIMEOUT 20)
    endforeach()
    add_executable(picodbg_peripheral_tests tests/test_peripherals.cpp)
    target_link_libraries(picodbg_peripheral_tests PRIVATE picodbg_core)
    add_test(NAME peripherals COMMAND picodbg_peripheral_tests)
    set_tests_properties(peripherals PROPERTIES TIMEOUT 15)
    if(TARGET pico-debugger)
        add_test(NAME qt_smoke COMMAND pico-debugger --demo --smoke-test)
        set_tests_properties(qt_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
        add_executable(picodbg_gui_tests tests/test_gui.cpp)
        target_link_libraries(picodbg_gui_tests PRIVATE picodbg_widgets)
        add_test(NAME qt_interactions COMMAND picodbg_gui_tests)
        set_tests_properties(qt_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    endif()
endif()
install(TARGETS pico-debugger-cli RUNTIME DESTINATION bin)
if(TARGET pico-debugger)
    install(TARGETS pico-debugger RUNTIME DESTINATION bin)
endif()
